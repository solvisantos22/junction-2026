This is an untested autopilot module; I have not run it against the simulator. It works in cycles of drive, stop, estimate position, correct and replan.

**Position estimate.** Each cycle sends a leg of drive commands, then waits until enough telemetry from after the last command has arrived. The rover is stationary by then, so a trimmed mean of those noisy position fixes gives its position.

**Calibration.** Each leg's predicted displacement is compared with the displacement the fixes show. The comparison updates a heading-bias estimate and a wheel-scale estimate.

**Routing.** A Dijkstra search over the 5 m vertex grid chooses the route. It rejects edges that are too steep or that would slip more than the 0.45 limit under pessimistic slip values. It penalises sand, crust regions and spots where a safety stop happened. It relaxes the limits if no target is reachable.

**Command generation.** Each command is generated by simulating `lightlag_surface.drive` with estimated slip values.

**Sampling.** When a target is within about 7 m, the next uplink carries a short correcting drive, then the sample, then the leg to the next target. Failed samples are retried at most twice.

**Choosing targets.** The next target is the one with the best value per tick, using the late value if the deadline would be missed. It must leave enough time to get back to the lander.

**Return.** If no target fits the remaining time, the rover heads for the lander and waits there, correcting its position if the fixes show it is more than 1.8 m from the lander.

```python
import math
import heapq

import lightlag_surface as LS

TERR = {"b": "bedrock", "r": "regolith", "s": "sand", "k": "rocks"}
DIRS = [(dx, dy) for dx in range(-2, 3) for dy in range(-2, 3)
        if (dx, dy) != (0, 0) and math.gcd(abs(dx), abs(dy)) == 1]
MAXBASE = {"bedrock": 0.05, "regolith": 0.12, "sand": 0.35, "rocks": 0.24}
MIDBASE = {"bedrock": 0.03, "regolith": 0.085, "sand": 0.25, "rocks": 0.18}
MAXGAIN = {"bedrock": 0.8, "regolith": 1.5, "sand": 4.5, "rocks": 1.2}
NOMGAIN = {"bedrock": 0.8, "regolith": 1.5, "sand": 3.5, "rocks": 1.2}


def trimmed(vals):
    v = sorted(vals)
    n = len(v)
    if n >= 4:
        v = v[1:-1]
        return sum(v) / len(v)
    if n == 3:
        return v[1]
    return sum(v) / n


def wrap180(a):
    while a > 180:
        a -= 360
    while a < -180:
        a += 360
    return a


class Pilot:
    def __init__(self, tools):
        self.tools = tools
        self.rules = tools.rules()
        self.mp = tools.map()
        mp = self.mp
        self.S = LS.Surface(mp["cell_m"], mp["elevation_cm"], mp["terrain"])
        self.phys = self.rules["physics"]
        self.cell = float(mp["cell_m"])
        self.nx, self.ny = self.S.nx, self.S.ny
        self.base = (float(mp["base"][0]), float(mp["base"][1]))
        self.maxt = self.rules["max_ticks"]
        self.crust = mp.get("crust_regions", [])
        self.mscale = 1.0
        self.bhat = 0.0
        self.bad = []
        self.done = set()
        self.pending = {}
        self.attempts = {}
        self.skip = set()
        self.ecache = {}
        self.pk = {}
        self.seen = set()
        self.battery = 100.0
        self.bat_tick = -1
        self.dead = False
        self.waiting = False
        self.ready = True
        self.stopped = False
        self.last_exec = 0
        self.sent_tick = 0
        self.idle_next = 0
        self.ncal = 0
        self.dj_cache = None

    # ---------- telemetry ----------
    def refresh(self):
        for pkt in self.tools.telemetry_log():
            t = pkt["tick"]
            if t in self.pk:
                continue
            self.pk[t] = pkt
            for s in pkt.get("sampled_targets", []):
                self.done.add(s)
            if t >= self.bat_tick:
                self.bat_tick = t
                self.battery = pkt["battery"]
            if not pkt.get("alive", True):
                self.dead = True
            for ev in pkt.get("events", []):
                if ev.get("type") == "safety_stop" and self.waiting and t >= self.sent_tick:
                    self.stopped = True
                    self.bad.append((pkt["position"][0], pkt["position"][1]))

    # ---------- graph ----------
    def in_crust(self, x, y):
        for c in self.crust:
            if c["x1"] <= x <= c["x2"] and c["y1"] <= y <= c["y2"]:
                return True
        return False

    def edge(self, i, j, k, relax):
        key = (i, j, k, relax)
        if key in self.ecache:
            return self.ecache[key]
        dx, dy = DIRS[k]
        c = self.cell
        x0, y0 = i * c, j * c
        L = math.hypot(dx, dy) * c
        n = max(1, int(math.ceil(L / 2.5)))
        ux, uy = dx * c / L, dy * c / L
        seg = L / n
        tl = 0.33 if relax else 0.29
        slim = 0.5 if relax else 0.42
        pts = [self.S.sample(x0 + ux * seg * m, y0 + uy * seg * m) for m in range(n + 1)]
        cost = 0.0
        wheel = 0.0
        res = None
        ok = True
        for m in range(n + 1):
            if math.hypot(pts[m][1], pts[m][2]) > tl:
                ok = False
                break
        if ok:
            for m in range(n):
                h, gx, gy, code = pts[m]
                h2 = pts[m + 1][0]
                name = TERR[code]
                grade = gx * ux + gy * uy
                cross = -gx * uy + gy * ux
                g = grade if grade > 0 else 0.0
                if MAXBASE[name] + MAXGAIN[name] * g + 0.02 > slim:
                    ok = False
                    break
                slip = MIDBASE[name] + NOMGAIN[name] * g
                slip = min(slip, 0.9)
                w = seg / (1 - slip)
                climb = max(0.0, h2 - h)
                dc = self.phys["drive_cost_per_wheel_m"][name]
                sd = self.phys["side_slip_gain"][name]
                cc = w * (1 + 3 * dc) + 6 * climb + seg * sd * abs(cross) * 4
                if name == "sand":
                    cc *= 1.4
                x = x0 + ux * seg * m
                y = y0 + uy * seg * m
                if self.crust and self.in_crust(x, y):
                    cc *= 1.5
                cost += cc
                wheel += w
            if ok:
                res = (cost, wheel)
        self.ecache[key] = res
        return res

    def node(self, x, y):
        i = min(self.nx - 1, max(0, int(round(x / self.cell))))
        j = min(self.ny - 1, max(0, int(round(y / self.cell))))
        return j * self.nx + i

    def dijk(self, s, relax):
        if self.dj_cache and self.dj_cache[0] == (s, relax, len(self.bad)):
            return self.dj_cache[1]
        N = self.nx * self.ny
        INF = float("inf")
        dist = [INF] * N
        Mw = [0.0] * N
        prev = [-1] * N
        dist[s] = 0.0
        heap = [(0.0, s)]
        nx, ny, c = self.nx, self.ny, self.cell
        bad = self.bad
        while heap:
            d, u = heapq.heappop(heap)
            if d > dist[u]:
                continue
            j, i = divmod(u, nx)
            for k, (dx, dy) in enumerate(DIRS):
                ni, nj = i + dx, j + dy
                if ni < 0 or nj < 0 or ni >= nx or nj >= ny:
                    continue
                e = self.edge(i, j, k, relax)
                if e is None:
                    continue
                cost, w = e
                if bad:
                    mx, my = (i + dx / 2) * c, (j + dy / 2) * c
                    for bx, by in bad:
                        if math.hypot(mx - bx, my - by) < 12:
                            cost *= 3
                            break
                v = nj * nx + ni
                nd = d + cost
                if nd < dist[v]:
                    dist[v] = nd
                    Mw[v] = Mw[u] + w
                    prev[v] = u
                    heapq.heappush(heap, (nd, v))
        out = (dist, Mw, prev)
        self.dj_cache = ((s, relax, len(self.bad)), out)
        return out

    def path_to(self, pos, target):
        s = self.node(*pos)
        tn = self.node(*target)
        for relax in (False, True):
            dist, Mw, prev = self.dijk(s, relax)
            if dist[tn] < float("inf"):
                nodes = []
                u = tn
                while u != -1:
                    nodes.append(u)
                    if u == s:
                        break
                    u = prev[u]
                nodes.reverse()
                pts = [((u % self.nx) * self.cell, (u // self.nx) * self.cell) for u in nodes]
                if len(pts) > 1 and math.hypot(pts[0][0] - pos[0], pts[0][1] - pos[1]) < 3:
                    pts = pts[1:]
                return pts + [target]
        return [target]

    # ---------- command generation ----------
    def sim(self, pos, hd, m):
        return LS.drive(self.S, self.phys, pos[0], pos[1], hd, m * self.mscale, MIDBASE_FULL)

    def gen_leg(self, pos, wps, cap, final_tol=0.6):
        cmds = []
        idx = 0
        stuck = 0
        while len(cmds) < cap and idx < len(wps):
            w = wps[idx]
            last = idx == len(wps) - 1
            d = math.hypot(w[0] - pos[0], w[1] - pos[1])
            tol = final_tol if last else 2.5
            if d <= tol:
                idx += 1
                continue
            hd = math.degrees(math.atan2(w[1] - pos[1], w[0] - pos[0]))
            pr = self.sim(pos, hd, 1.0)
            ratio = pr["ground"] if not pr["stop"] else 0.7
            ratio = max(0.25, ratio)
            m = min(5.0, d / ratio)
            m = max(m, 0.3)
            res = self.sim(pos, hd, m)
            if res["stop"]:
                break
            npos = (res["x"], res["y"])
            nd = math.hypot(w[0] - npos[0], w[1] - npos[1])
            if nd > d - 0.05:
                stuck += 1
                if stuck > 2:
                    break
            cmds.append({"action": "drive", "heading_deg": round((hd - self.bhat) % 360.0, 3),
                         "metres": round(m, 3)})
            pos = npos
        return cmds, pos, idx >= len(wps)

    # ---------- selection ----------
    def select(self, pos, t_next, exclude):
        if self.battery < 18:
            return None
        targets = self.tools.map()["targets"]
        cands = [t for t in targets if t["id"] not in self.done and t["id"] not in exclude
                 and t["id"] not in self.skip and self.attempts.get(t["id"], 0) < 2
                 and t["id"] not in self.pending]
        if not cands:
            return None
        s = self.node(*pos)
        best = None
        bestrate = 0
        for relax in (False, True):
            dist, Mw, prev = self.dijk(s, relax)
            reach = 0
            for t in cands:
                tn = self.node(t["x"], t["y"])
                if dist[tn] == float("inf"):
                    continue
                reach += 1
                dd = math.hypot(pos[0] - t["x"], pos[1] - t["y"])
                if dd <= 7:
                    M = dd / 0.85 / self.mscale
                else:
                    nxp = (tn % self.nx) * self.cell
                    nyp = (tn // self.nx) * self.cell
                    M = Mw[tn] / self.mscale + math.hypot(nxp - t["x"], nyp - t["y"]) / 0.85 / self.mscale
                trav = M / 4.5
                arrival = t_next + trav
                t_samp = arrival + (2 if dd <= 7 else 12)
                ret = math.hypot(t["x"] - self.base[0], t["y"] - self.base[1]) * 1.35 / 4.4
                if t_samp + 2 + ret > 400:
                    continue
                dl = t.get("deadline_tick")
                v = t["value"] if (dl is None or t_samp <= dl) else t["late_value"]
                if v <= 0:
                    continue
                rate = v / (trav + 12)
                if rate > bestrate:
                    bestrate = rate
                    best = t
            if reach > 0:
                break
        return best

    # ---------- planning ----------
    def plan(self, tick):
        tr = tick if tick % 8 < 4 else tick + 8 - tick % 8
        t0 = tr + 4
        cmds = []
        pend = []
        pos = self.pos
        chained = set()
        cap_far = 6 if self.ncal < 2 else 14
        for it in range(3):
            t_next = t0 + len(cmds)
            t = self.select(pos, t_next, chained)
            if t is None:
                d = math.hypot(pos[0] - self.base[0], pos[1] - self.base[1])
                if d > 1.8:
                    if d <= 7:
                        wps = [self.base]
                    else:
                        wps = self.path_to(pos, self.base)
                    c, pos2, ok = self.gen_leg(pos, wps, cap_far)
                    cmds += c
                    pos = pos2
                break
            tx, ty = t["x"], t["y"]
            d = math.hypot(pos[0] - tx, pos[1] - ty)
            if d <= 7:
                c, pos2, ok = self.gen_leg(pos, [(tx, ty)], 4)
                if not c and d > 2.3:
                    self.skip.add(t["id"])
                    continue
                cmds += c
                cmds.append({"action": "sample"})
                pend.append((t["id"], len(cmds) - 1))
                chained.add(t["id"])
                pos = pos2
                continue
            wps = self.path_to(pos, (tx, ty))
            c, pos2, ok = self.gen_leg(pos, wps, cap_far)
            if not c:
                self.skip.add(t["id"])
                continue
            cmds += c
            pos = pos2
            break
        return cmds, pend, pos

    def act(self, tick):
        cmds, pend, pred_end = self.plan(tick)
        if not cmds:
            self.idle_next = tick + 4
            return
        execs = []
        for i in range(0, len(cmds), 5):
            chunk = cmds[i:i + 5]
            try:
                acks = self.tools.uplink(chunk)
            except Exception:
                return
            for a in acks:
                execs.append(a["execute_tick"])
        self.last_exec = max(execs)
        for tid, idx in pend:
            if idx < len(execs):
                self.pending[tid] = execs[idx]
        self.waiting = True
        self.ready = False
        self.prev_anchor = self.pos
        self.pred_end = pred_end
        self.stopped = False
        self.sent_tick = tick

    def check_ready(self, tick):
        fixes = [self.pk[t]["position"] for t in sorted(self.pk) if t >= self.last_exec]
        n = len(fixes)
        age = tick - self.last_exec
        if not (n >= 4 or (n >= 2 and age >= 12) or (n >= 1 and age >= 20)):
            return
        fx = fixes[-8:]
        newpos = (trimmed([f[0] for f in fx]), trimmed([f[1] for f in fx]))
        pa = self.prev_anchor
        A = (newpos[0] - pa[0], newpos[1] - pa[1])
        P = (self.pred_end[0] - pa[0], self.pred_end[1] - pa[1])
        LA, LP = math.hypot(*A), math.hypot(*P)
        if not self.stopped and LP > 15 and LA > 5:
            dlt = wrap180(math.degrees(math.atan2(A[1], A[0]) - math.atan2(P[1], P[0])))
            w = min(0.8, LP / 40.0)
            self.bhat = max(-12.0, min(12.0, self.bhat + w * dlt))
            r = max(0.6, min(1.5, LA / LP))
            self.mscale = max(0.75, min(1.25, self.mscale * (r ** w)))
            self.ncal += 1
        for tid in list(self.pending):
            if tid in self.done:
                pass
            else:
                self.attempts[tid] = self.attempts.get(tid, 0) + 1
            del self.pending[tid]
        self.pos = newpos
        self.waiting = False
        self.ready = True

    def run(self):
        tools = self.tools
        tick = 0
        self.refresh()
        p0 = self.pk[0]["position"] if 0 in self.pk else self.base
        self.pos = (float(p0[0]), float(p0[1]))
        while True:
            if self.dead:
                break
            if self.ready and tick >= self.idle_next:
                try:
                    self.act(tick)
                except Exception:
                    self.idle_next = tick + 4
            r = tools.advance(1)
            tick = r["current_tick"]
            self.refresh()
            if r["episode_complete"]:
                break
            if self.waiting:
                self.check_ready(tick)


MIDBASE_FULL = dict(MIDBASE)


def play(tools):
    pilot = Pilot(tools)
    try:
        pilot.run()
    except Exception:
        # keep the clock running to the end so the mission completes
        try:
            while True:
                r = tools.advance(5)
                if r["episode_complete"]:
                    break
        except Exception:
            pass
```
